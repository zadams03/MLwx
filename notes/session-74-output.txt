  """Session 74 Step 2 helper (read-only, no fit): build the record script's
  RNO train/test matrices with its own functions, and session 73's with its
  own functions, and compare them bit for bit. Writes nothing."""
  import sys
  sys.dont_write_bytecode = True
  ROOT = "/Users/zacharyadams/Coding Projects/MLwx"
  sys.path.insert(0, ROOT + "/scripts")
  
  import session62_reserved_confirm as rec   # frozen; imported, never run
  import session73_fit_score as s73
  import numpy as np
  
  # ---- record side (run_confirm's own steps, RNO only, no fit) ----
  l_all, hl = rec.load_family("L", ["lapse_rate_t2_t850"])
  d_all, hd = rec.load_family("D", ["dewpoint_depression_t2m"])
  t_all, ht = rec.load_family("T", ["pressure_tendency_3h_hpa"])
  r_all, hr = rec.load_family("R", ["dswrf_2h_wm2"])
  assert hl == hd == ht == hr == 0
  base_all = rec.load_base_unfiltered()
  merged = rec.build_complete_case(base_all, l_all, d_all, t_all, r_all)
  cf = rec.CONFIRMATION_FOLD
  obs_all, _, _ = rec.load_obs_all("RNO", rec.AIRPORTS["RNO"])
  tr_raw = {d: r for d, r in merged["RNO"].items() if cf["train_start"] <= d <= cf["train_end"]}
  te_raw = {d: r for d, r in merged["RNO"].items() if cf["test_start"] <= d <= cf["test_end"]}
  rtr, _ = rec.join_obs("RNO", tr_raw, obs_all)
  rte, _ = rec.join_obs("RNO", te_raw, obs_all)
  print("record FINAL_FEATURE_KEYS:", rec.FINAL_FEATURE_KEYS)
  print("record BASE_KEYS        :", rec.BASE_KEYS)
  RX_tr = rec.features_matrix(rtr, rec.FINAL_FEATURE_KEYS)
  RX_te = rec.features_matrix(rte, rec.FINAL_FEATURE_KEYS)
  Ry = np.array([r["resid"] for r in rtr], dtype=float)
  RB_tr = rec.features_matrix(rtr, rec.BASE_KEYS)
  print("record X dtype/type:", RX_tr.dtype, type(RX_tr).__name__, "y dtype:", Ry.dtype)
  print("record LGB_PARAMS == s73 PARAMS:", rec.LGB_PARAMS == s73.PARAMS)
  
  # ---- session 73 side (its own functions, Variant P) ----
  rows, counts, nonfinite = s73.load_tables()
  feats, mask = s73.model_columns("BDLRT_P")
  train, _ = s73.complete_case(rows, "train", mask)
  test, _ = s73.complete_case(rows, "test", mask)
  SX_tr = np.array([[r[c] for c in feats] for r in train], dtype=np.float64)
  SX_te = np.array([[r[c] for c in feats] for r in test], dtype=np.float64)
  Sy = np.array([r["residual_c"] for r in train], dtype=np.float64)
  print("s73 feature order:", feats)
  
  # ---- compare ----
  print("n_train record/s73:", len(rtr), len(train), " n_test:", len(rte), len(test))
  print("train dates identical and in same order:",
        [r["date"].isoformat() for r in rtr] == [r["target_date"] for r in train])
  print("test dates identical and in same order :",
        [r["date"].isoformat() for r in rte] == [r["target_date"] for r in test])
  print("train dates ascending (record, s73):",
        all(rtr[i]["date"] < rtr[i+1]["date"] for i in range(len(rtr)-1)),
        all(train[i]["target_date"] < train[i+1]["target_date"] for i in range(len(train)-1)))
  perm = [rec.FINAL_FEATURE_KEYS.index(k) for k in
          ["temp", "season_sin", "season_cos", "cloud_cover", "wind_speed_10m",
           "lapse_rate_t2_t850", "dewpoint_depression_t2m_floored",
           "pressure_tendency_3h_hpa", "dswrf_2h_wm2"]]
  print("column permutation record->s73:", perm)
  for name, R, S in [("X_train", RX_tr[:, perm], SX_tr), ("X_test", RX_te[:, perm], SX_te)]:
      eq = (R == S)
      print("%s bit-equal after permutation: %s  (per column equal counts: %s)"
            % (name, bool(eq.all()), eq.sum(axis=0).tolist()))
  print("X_train unpermuted equal (order matters):", bool((RX_tr == SX_tr).all()))
  print("B X_train equal:", bool((RB_tr == SX_tr[:, :5]).all()))
  ydiff = np.abs(Ry - Sy)
  print("target: bit-equal rows %d of %d; max abs diff %r; round(record,3)==s73 on %d"
        % (int((Ry == Sy).sum()), len(Ry), float(ydiff.max()),
           int(sum(round(a, 3) == b for a, b in zip(Ry, Sy)))))
  print("test obs equal:", all(a["obs"] == b["obs_c"] for a, b in zip(rte, test)),
        " test fc equal:", all(a["fc"] == b["forecast_c"] for a, b in zip(rte, test)))
  Su = np.array([r["obs_c"] - r["forecast_c"] for r in train], dtype=np.float64)
  print("unrounded target from s72 (obs_c - forecast_c) bit-equal to record resid: %d of %d"
        % (int((Su == Ry).sum()), len(Ry)))
